% A resume, for a job application.
%
% On the article class rather than a CV class, so it typesets on the TeX
% the installer sets up with the same packages every other template uses.
% The whole layout is four commands defined below; there is nothing to
% learn and nothing that stops you changing it.  The cover letter beside
% this file starts from the same name and contact line, so change both.
%
% One page is the rule for most listings.  The page follows as you type,
% so you can see when it spills.

\documentclass[11pt]{article}

\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage[margin=0.85in]{geometry}
\usepackage{parskip}
\usepackage{microtype}
\usepackage{enumitem}
\usepackage{hyperref}
\hypersetup{colorlinks, urlcolor=blue}

\pagestyle{empty}
\setlist[itemize]{leftmargin=1.2em, nosep, itemsep=2pt}

% A heading with a rule under it.
\newcommand{\heading}[1]{%
  \vspace{6pt}{\large\bfseries #1}\\[-9pt]\rule{\linewidth}{0.4pt}\par}
% What, when, where and in what role: two lines, dates and places at the right.
\newcommand{\entry}[4]{%
  \textbf{#1}\hfill #2\\\emph{#3}\hfill #4\par}
% A publication: authors, title, venue and year, hanging.
\newcommand{\paper}[1]{\hangindent=1.2em #1\par\vspace{2pt}}

\begin{document}
